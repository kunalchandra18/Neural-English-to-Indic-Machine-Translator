# Dependencies for the hosted Streamlit demo.
# Streamlit Community Cloud installs this file from the repository root.
# Training dependencies live in requirements-train.txt.

# The +cpu build keeps the install near 200 MB; the default PyPI wheel bundles CUDA
# and will not fit in a free hosting tier. Pin the local version, not an exact release:
# PyTorch retires old wheels, and an exact pin breaks the build once it is dropped.
# A local version sorts above the same base version (PEP 440), so the +cpu wheel wins
# over the CUDA wheel PyPI serves under the same number.
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.9,<3

numpy
PyYAML
streamlit>=1.36
huggingface_hub>=0.24,<1.0
# spaCy 3.7 has no wheels past cp312. On the 3.14 runner pip fell back to building
# from source, and its blis 0.7 dependency pins Cython <3.0 while NumPy's Cython
# headers require >=3.0, so the build cannot succeed. 3.8.10 is the first release
# with cp314 wheels, and it pulls thinc 8.3.x / blis 1.3.x, which have them too.
# The model wheel must track spaCy's minor version.
spacy>=3.8.10,<3.9
en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.8.0/en_core_web_sm-3.8.0-py3-none-any.whl
